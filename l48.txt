import tensorflow as tf
from tensorflow.keras import Sequential
from tensorflow.keras.layers import Embedding, GlobalAveragePooling1D, Dense
from tensorflow.keras.utils import pad_sequences

(x_train, y_train), (x_test, y_test) = tf.keras.datasets.imdb.load_data(num_words=10000)
x_train, x_test = pad_sequences(x_train, maxlen=200), pad_sequences(x_test, maxlen=200)
model = Sequential([Embedding(10000, 32), GlobalAveragePooling1D(), Dense(64, activation='relu'), Dense(1, activation='sigmoid')])
model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])
model.fit(x_train, y_train, epochs=5, batch_size=32, validation_split=0.2)
print("Test Accuracy:", model.evaluate(x_test, y_test)[1])
predictions = model.predict(x_test[:5])
print("\nReview Classification:")
for i, p in enumerate(predictions[:, 0]):
    print("Review", i + 1, ":", "Positive" if p >= 0.5 else "Negative")
    print("Probability:", p)

import numpy as np
import matplotlib.pyplot as plt
from numpy.lib.stride_tricks import sliding_window_view
from sklearn.preprocessing import MinMaxScaler
from tensorflow.keras import Sequential
from tensorflow.keras.layers import LSTM, Dense

time = np.arange(0, 200, 0.1)
data = np.sin(time) + 0.1 * np.random.randn(len(time))
plt.plot(time, data)
plt.title("Synthetic Time Series Data")
plt.show()
scaler = MinMaxScaler()
w = sliding_window_view(scaler.fit_transform(data.reshape(-1, 1)).ravel(), 21)
X, y = w[:, :20, None], w[:, 20]
n = int(len(X) * 0.8)
model = Sequential([LSTM(50, activation='relu', input_shape=(20, 1)), Dense(1)])
model.compile(optimizer='adam', loss='mse')
h = model.fit(X[:n], y[:n], epochs=10, batch_size=32, validation_data=(X[n:], y[n:]), verbose=1)
plt.plot(h.history['loss'], label='Train Loss')
plt.plot(h.history['val_loss'], label='Validation Loss')
plt.legend()
plt.show()
plt.plot(scaler.inverse_transform(y[n:].reshape(-1, 1)), label='Actual')
plt.plot(scaler.inverse_transform(model.predict(X[n:])), label='Predicted')
plt.legend()
plt.show()

import tensorflow as tf
import matplotlib.pyplot as plt
import numpy as np
from tensorflow.keras import layers

(x_train, y_train), (x_test, y_test) = tf.keras.datasets.cifar10.load_data()
y_train, y_test = y_train.flatten(), y_test.flatten()
x_train, x_test = tf.image.resize(x_train, (96, 96)), tf.image.resize(x_test, (96, 96))
base = tf.keras.applications.MobileNetV2(input_shape=(96, 96, 3), include_top=False, weights="imagenet")
base.trainable = False
model = tf.keras.Sequential([base, layers.GlobalAveragePooling2D(), layers.Dense(64, activation="relu"), layers.Dense(10, activation="softmax")])
model.compile(optimizer="adam", loss="sparse_categorical_crossentropy", metrics=["accuracy"])
h = model.fit(x_train, y_train, epochs=5, batch_size=32, validation_split=0.2)
acc = model.evaluate(x_test, y_test)[1]
print("Test Accuracy:", acc * 100, "%")
classes = "airplane automobile bird cat deer dog frog horse ship truck".split()
print("Actual:", classes[y_test[0]])
print("Predicted:", classes[np.argmax(model.predict(x_test[:1]))])
print("Training Accuracy:", h.history["accuracy"][-1] * 100, "%")
print("Validation Accuracy:", h.history["val_accuracy"][-1] * 100, "%")
print("Test Accuracy:", acc * 100, "%")
plt.figure(figsize=(8, 5))
plt.plot(h.history["accuracy"], label="Training Accuracy")
plt.plot(h.history["val_accuracy"], label="Validation Accuracy")
plt.xlabel("Epoch")
plt.ylabel("Accuracy")
plt.title("Training and Validation Accuracy")
plt.legend()
plt.grid()
plt.show()

from sklearn.model_selection import train_test_split
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.linear_model import LogisticRegression
from sklearn.pipeline import make_pipeline

reviews = ["I loved this movie, it was fantastic", "The product is excellent and very useful", "Amazing experience, I really enjoyed it", "The service was wonderful", "Very happy with my purchase", "This product is terrible", "I hated this movie", "Very disappointing experience", "The product broke after one day", "Poor quality and bad service"]
labels = [1] * 5 + [0] * 5
X_train, _, y_train, _ = train_test_split(reviews, labels, test_size=0.2, random_state=42, stratify=labels)
model = make_pipeline(TfidfVectorizer(), LogisticRegression()).fit(X_train, y_train)
new_reviews = ["The movie was amazing and I loved it", "The product was awful and disappointing", "I am very happy with this purchase"]
for r, p in zip(new_reviews, model.predict(new_reviews)):
    print("\nReview:", r)
    print("Sentiment:", "Positive" if p == 1 else "Negative")
